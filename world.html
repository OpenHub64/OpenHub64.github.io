// OpenHub World — authoritative server, step 3: health & combat. // // Steps 1 (auth) and 2 (movement) are unchanged in spirit below. New in // this step: the server owns health and bullets too. A client can say "I // shot" and "here's my aim angle" — it can never say "I hit someone" or // "my health is X." Hit detection, damage, death, and regen all run here, // every tick, for every player, whether or not their tab is open. This is // the piece that makes a hit land even if the target backgrounded their // tab — their physics + health keep ticking on the server regardless. import 'dotenv/config'; import { WebSocketServer } from 'ws'; import { createClient } from '@supabase/supabase-js'; import { createInitialState, stepPlayer, createBullet, stepBullets, MAX_HEALTH, BULLET_DAMAGE, RESPAWN_MS, SHOOT_COOLDOWN_MS, REGEN_DELAY_MS, REGEN_PER_SEC } from './physics.js'; const SUPABASE_URL = process.env.SUPABASE_URL; const SUPABASE_SERVICE_ROLE_KEY = process.env.SUPABASE_SERVICE_ROLE_KEY; const PORT = Number(process.env.PORT) || 8787; if (!SUPABASE_URL || !SUPABASE_SERVICE_ROLE_KEY) { console.error( 'Missing SUPABASE_URL or SUPABASE_SERVICE_ROLE_KEY.\n' + 'Copy .env.example to .env and fill both in before starting the server.' ); process.exit(1); } // Service-role client: bypasses Row Level Security. This is what lets the // server write world_players/player_stats on everyone's behalf. It must // only ever be constructed here, server-side. const supabase = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY, { auth: { persistSession: false } }); // userId -> { // socket, username, color, // physics: { x, y, vy, moveVel, onGround, jumpsUsed, jumpKeyHeld }, // input: { left, right, jump }, // aimAngle, // health, alive, respawnAt, lastDamageAt, lastShotAt // } const players = new Map(); // Global bullet pool, id -> { id, ownerId, x, y, vx, vy }. Not per-player — // a bullet can hit anyone, so it lives at the world level. const bullets = new Map(); const wss = new WebSocketServer({ port: PORT }); console.log(`OpenHub World server listening on ws://localhost:${PORT}`); function colorForSeed(seed) { const colors = ['#6ee7c8', '#ff6b6b', '#ffd166', '#6ea8ff', '#c792ea', '#ff9f6e', '#7ee787']; let h = 0; for (let i = 0; i < seed.length; i++) h = (h * 31 + seed.charCodeAt(i)) >>> 0; return colors[h % colors.length]; } wss.on('connection', (socket) => { let userId = null; console.log('Client connected, awaiting auth…'); const authTimeout = setTimeout(() => { if (!userId) { console.log('Client failed to auth in time, closing.'); socket.close(); } }, 8000); socket.on('message', async (raw) => { let msg; try { msg = JSON.parse(raw.toString()); } catch { return; // ignore anything that isn't valid JSON } if (msg.type === 'auth') { if (userId) return; // already authenticated on this socket const { data, error } = await supabase.auth.getUser(msg.token); if (error || !data?.user) { socket.send(JSON.stringify({ type: 'authError', message: 'Invalid session' })); socket.close(); return; } userId = data.user.id; clearTimeout(authTimeout); const { data: profile } = await supabase .from('profiles') .select('username') .eq('id', userId) .single(); const username = profile?.username || 'Anon'; players.set(userId, { socket, username, color: colorForSeed(userId), physics: createInitialState(), input: { left: false, right: false, jump: false }, aimAngle: 0, health: MAX_HEALTH, alive: true, respawnAt: 0, lastDamageAt: -Infinity, lastShotAt: -Infinity, lastInputSeq: 0 }); console.log(`Authenticated: ${username} (${userId}) — ${players.size} online`); socket.send(JSON.stringify({ type: 'authOk', username })); broadcastPlayerCount(); return; } if (!userId) return; // everything below requires auth const p = players.get(userId); if (!p) return; if (msg.type === 'input') { // Trust nothing but the shape — a client can say what keys it's // holding and where it's aiming, never where it actually is. p.input = { left: !!msg.left, right: !!msg.right, jump: !!msg.jump }; if (typeof msg.aimAngle === 'number' && isFinite(msg.aimAngle)) { p.aimAngle = msg.aimAngle; } // Client tags each fixed-step input with a sequence number so it can // later tell which of its locally-predicted steps this server tick // has already accounted for (see lastInputSeq in the outgoing // snapshot, and the client's reconcile-then-replay logic). if (typeof msg.seq === 'number' && isFinite(msg.seq)) { p.lastInputSeq = msg.seq; } return; } if (msg.type === 'shoot') { if (!p.alive) return; const now = Date.now(); if (now - p.lastShotAt < SHOOT_COOLDOWN_MS) return; // rate-limited server-side, not just client-side p.lastShotAt = now; const b = createBullet(userId, p.physics.x, p.physics.y, p.aimAngle); bullets.set(b.id, b); return; } }); socket.on('close', () => { clearTimeout(authTimeout); if (userId && players.has(userId)) { const { username } = players.get(userId); players.delete(userId); console.log(`Disconnected: ${username} (${userId}) — ${players.size} online`); broadcastPlayerCount(); } }); socket.on('error', (err) => { console.error('Socket error:', err.message); }); }); function broadcastPlayerCount() { broadcastAll({ type: 'playerCount', count: players.size }); } function broadcastAll(payload) { const raw = JSON.stringify(payload); for (const { socket } of players.values()) { if (socket.readyState === socket.OPEN) socket.send(raw); } } // Fire-and-forget — a stat write failing shouldn't interrupt the game. // Uses server-only RPCs that take an explicit player id (unlike the // client-facing ones, which infer the id from the caller's own session — // the server has no "own session," it's acting on players' behalf, so it // needs the explicit-id versions. See server-stats.sql. async function recordKill(id, username) { try { await supabase.rpc('server_increment_kills', { p_id: id, p_username: username }); } catch (err) { console.error('recordKill failed:', err.message); } } async function recordDeath(id, username) { try { await supabase.rpc('server_increment_deaths', { p_id: id, p_username: username }); } catch (err) { console.error('recordDeath failed:', err.message); } } // ── Tick loop ──────────────────────────────────────────────────────────── // This MUST be a true fixed timestep, not "measured elapsed time capped at // 50ms." The client predicts its own movement locally and replays any // unacknowledged inputs after each server snapshot by stepping the exact // same physics with dt hardcoded to FIXED_DT (see world.html — it never // measures real time for this, on purpose). If the server instead steps // with whatever real time happened to elapse between two setInterval // callbacks (which is never exactly 50.000ms — timer jitter is normally a // few ms either way), client and server are integrating the same equations // with slightly different dt every tick. That's not cosmetic: stepping // x += v*dt with dt=0.0498 vs dt=0.0500 lands at a different x, which // right at a platform edge can be the difference between landing and not. // The result is the server "disagreeing" with the client's prediction on // almost every packet, which is exactly the constant-correction jitter this // fixes. The standard answer (this is the "Fix Your Timestep" pattern used // by Minecraft and most other tick-based multiplayer games) is: decouple // the simulation step size from wall-clock time entirely. Accumulate real // elapsed time, then run as many WHOLE fixed-size steps as that time // covers — each step's dt is always exactly the same constant, so a given // input sequence always produces bit-identical results, however irregular // the actual callback timing was. const TICK_HZ = 20; const FIXED_DT = 1 / TICK_HZ; // exactly 0.05 — must equal world.html's FIXED_DT const TICK_MS = 1000 * FIXED_DT; // driver interval; NOT the simulation step size const MAX_CATCHUP_STEPS = 8; // mirrors the client's own stall guard (FIXED_DT * 8) let lastTick = Date.now(); let accumulator = 0; setInterval(() => { const now = Date.now(); accumulator += (now - lastTick) / 1000; lastTick = now; if (players.size === 0) { accumulator = 0; // don't let idle time pile up into a burst of steps later return; } accumulator = Math.min(accumulator, FIXED_DT * MAX_CATCHUP_STEPS); const outEvents = []; let stepsRun = 0; while (accumulator >= FIXED_DT) { accumulator -= FIXED_DT; stepsRun++; // Movement + respawn + regen, one fixed step. for (const p of players.values()) { if (!p.alive) { if (now >= p.respawnAt) { p.physics = createInitialState(); p.health = MAX_HEALTH; p.alive = true; p.lastDamageAt = -Infinity; } continue; } stepPlayer(p.physics, p.input, FIXED_DT, p.health); if (p.health < MAX_HEALTH && (now - p.lastDamageAt) > REGEN_DELAY_MS) { p.health = Math.min(MAX_HEALTH, p.health + REGEN_PER_SEC * FIXED_DT); } } // Bullets vs. platforms/players, one fixed step. const playersById = new Map(); for (const [id, p] of players.entries()) { playersById.set(id, { x: p.physics.x, y: p.physics.y, alive: p.alive }); } const hitEvents = stepBullets(bullets, playersById, FIXED_DT); for (const ev of hitEvents) { if (ev.kind === 'platform') { outEvents.push({ kind: 'impact', x: ev.x, y: ev.y }); continue; } if (ev.kind === 'player') { const target = players.get(ev.targetId); if (!target || !target.alive) continue; target.health = Math.max(0, target.health - BULLET_DAMAGE); target.lastDamageAt = now; const shooterId = ev.shooterId; outEvents.push({ kind: 'hit', x: ev.x, y: ev.y, targetId: ev.targetId, shooterId, damage: BULLET_DAMAGE }); if (target.health <= 0) { target.alive = false; target.respawnAt = now + RESPAWN_MS; outEvents.push({ kind: 'kill', targetId: ev.targetId, shooterId }); recordDeath(ev.targetId, target.username); if (shooterId && players.has(shooterId)) { recordKill(shooterId, players.get(shooterId).username); } } } } } if (stepsRun === 0) return; // nothing to report this callback const snapshot = []; for (const [id, p] of players.entries()) { snapshot.push({ id, username: p.username, color: p.color, x: Math.round(p.physics.x * 100) / 100, y: Math.round(p.physics.y * 100) / 100, vy: Math.round(p.physics.vy * 100) / 100, moveVel: Math.round(p.physics.moveVel * 100) / 100, onGround: p.physics.onGround, jumpsUsed: p.physics.jumpsUsed, jumpKeyHeld: p.physics.jumpKeyHeld, lastInputSeq: p.lastInputSeq, aimAngle: Math.round(p.aimAngle * 1000) / 1000, health: Math.round(p.health), alive: p.alive, respawnInMs: p.alive ? 0 : Math.max(0, p.respawnAt - now) }); } const bulletSnapshot = []; for (const b of bullets.values()) { bulletSnapshot.push({ id: b.id, ownerId: b.ownerId, x: Math.round(b.x), y: Math.round(b.y) }); } broadcastAll({ type: 'state', players: snapshot, bullets: bulletSnapshot, events: outEvents }); }, TICK_MS); // ── Periodic durable sync ─────────────────────────────────────────────── // Mirrors in-memory state into world_players so it survives a reconnect // or a server restart, and so it's visible even to tabs that never // connect to this server directly (e.g. while the game client is still // only Realtime-based). const DB_SYNC_MS = 2000; setInterval(async () => { for (const [id, p] of players.entries()) { try { await supabase.from('world_players').upsert({ id, username: p.username, color: p.color, x: p.physics.x, y: p.physics.y, health: Math.round(p.health), alive: p.alive, updated_at: new Date().toISOString() }); } catch (err) { console.error(`DB sync failed for ${p.username}:`, err.message); } } }, DB_SYNC_MS); process.on('SIGINT', () => { console.log('\nShutting down…'); wss.close(() => process.exit(0)); });
